# Third-party packages the PC app needs at runtime.
#
# The repository had no dependency manifest: the portable release bundles its
# own Python with these already installed, and development installs them by
# hand.  That worked while every dependency was optional-ish -- a missing
# camera library fails loudly at the point of use.  It stopped working when
# device pairing arrived, because pairing silently degrades to unavailable if
# its dependency is absent, and a release that cannot pair cannot accept input
# from a phone once enforcement is on.
#
# Versions here match what the 2.0 portable bundle actually ships, so this file
# describes reality rather than an aspiration.  Verify with:
#
#   python tools/check_runtime_deps.py
#   python tools/check_runtime_deps.py --bundle "build/release/.../python"

# Device pairing: ECDH P-256, HKDF and constant-time comparison for /ws/input
# authentication.  Not optional in a release build -- see device_pairing.py.
cryptography>=42.0

# Camera capture and pose inference.
opencv-python>=4.9
numpy>=1.26
mediapipe>=0.10

# Local microphone capture and offline speech recognition.
sounddevice>=0.4.6
vosk>=0.3.45
